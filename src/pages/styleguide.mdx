## 代码

```ts
export function readingTime(text: string): number {
  const chars = text.replace(/\s+/g, '').length;
  return Math.max(1, Math.round(chars / 400));
}
```

超过 24 行的代码块会自动收起，底部渐隐，按钮写着总行数。这里放一段真实尺寸的代码来验证效果：

```ts
const cache = new Map<string, Post>();

/** 命中缓存直接返回，避免重复过滤整张表 */
export async function getPost(slug: string): Promise<Post | null> {
  const hit = cache.get(slug);
  if (hit) return hit;

  const posts = await getPosts();
  const found = posts.find((post) => post.id === slug) ?? null;
  if (found) cache.set(slug, found);
  return found;
}

/** 时间上的前后篇，系列内的顺序另算 */
export async function getNeighbours(slug: string) {
  const posts = await getPosts();
  const index = posts.findIndex((post) => post.id === slug);

  if (index < 0) return { prev: null, next: null };

  return {
    prev: posts[index - 1] ?? null,
    next: posts[index + 1] ?? null,
  };
}

/** 标签频次，侧栏与搜索页的热门标签都取自这里 */
export function countTags(posts: Post[]): Array<[string, number]> {
  const counts = new Map<string, number>();
  for (const post of posts) {
    for (const tag of post.tags) {
      counts.set(tag, (counts.get(tag) ?? 0) + 1);
    }
  }
  return [...counts.entries()].sort((a, b) => b[1] - a[1]);
}
```

## 要点框

<KeyPoints>
  - 提示框只放「离开主线也不影响理解」的内容
  - 一篇长文里同类提示框不要超过三个
  - 警告类必须写清后果，而不是只写「注意」
</KeyPoints>

## 怎么用

文章写成 `.mdx` 后，这五个组件已经全局注册，不用在开头 import：

```mdx
<Callout type="warn" title="注意">
  内容里照样可以写 **Markdown**。
</Callout>

<KeyPoints>
  - 要点一
  - 要点二
</KeyPoints>

<Steps>
  <Step title="第一步">先做这个。</Step>
  <Step title="第二步">再做那个。</Step>
</Steps>

<Details title="展开看细节">
  可选内容放在这里。
</Details>

<Figure src="/og.png" alt="说明" caption="图注" width={1200} height={630} />
```

普通 `.md` 文件不支持组件。需要图注就把文章改成 `.mdx` 用 `<Figure>`；Markdown 原生图片只有 alt 和浏览器 tooltip，不会生成图注。
